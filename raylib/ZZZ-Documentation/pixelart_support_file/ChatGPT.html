Sì, posso aiutarti a programmare in C99 usando la libreria raylib 🙂

Raylib è una libreria molto accessibile per grafica 2D/3D, input, audio e altro, perfetta per progetti semplici o per imparare game development in C.

Per darti un’idea, ecco un esempio base (una finestra con uno sfondo e del testo):

C
#include "raylib.h"

int main(void)
{
    const int screenWidth = 800;
    const int screenHeight = 450;

    InitWindow(screenWidth, screenHeight, "Hello Raylib");

    SetTargetFPS(60);

    while (!WindowShouldClose())
    {
        BeginDrawing();

        ClearBackground(RAYWHITE);
        DrawText("Ciao dal mondo Raylib!", 190, 200, 20, DARKGRAY);

        EndDrawing();
    }

    CloseWindow();

    return 0;
}

Con raylib puoi fare facilmente:

  • gestione audio

  • semplici giochi

Raylib è nei repository ufficiali, quindi basta:

Bash
sudo pacman -S raylib

Questo ti installa libreria + header.

Compilazione (C99)

Per compilare un file main.c:

Bash
gcc main.c -o gioco -lraylib -lm -lpthread -ldl -lrt -lX11

Su Arch spesso basta anche:

Bash
gcc main.c -o gioco -lraylib

(se tutto è linkato correttamente)

Puoi testare con questo mini esempio:

C
#include "raylib.h"

int main(void)

    {

        ClearBackground(BLACK);
        DrawCircle(400, 300, 50, RED);
        EndDrawing();
    }
    CloseWindow();
    return 0;
}

🔹 1. Leggere i file .pix (POSIX su Arch)

Su Arch Linux puoi usare dirent.h:

C
#include <dirent.h>
#include <string.h>
#include <stdio.h>

#define MAX_FILES 256
#define MAX_NAME 256

int load_pix_files(const char *path, char files[MAX_FILES][MAX_NAME]) {
    DIR *dir = opendir(path);
    if (!dir) {
        perror("opendir");
        return 0;
    }

    struct dirent *entry;
    int count = 0;

    while ((entry = readdir(dir)) != NULL && count < MAX_FILES) {
        if (entry->d_type == DT_REG) { // file normale
            const char *name = entry->d_name;
            const char *ext = strrchr(name, '.');

            if (ext && strcmp(ext, ".pix") == 0) {
                strncpy(files[count], name, MAX_NAME);
                count++;
            }
        }
    }

    closedir(dir);
    return count;
}

🔹 2. Mostrare la lista con raylib

Qui un esempio con selezione via tastiera:

C
#include "raylib.h"

int main(void)
{
    const int screenWidth = 600;
    const int screenHeight = 400;

    InitWindow(screenWidth, screenHeight, "Selettore file .pix");

    char files[MAX_FILES][MAX_NAME];
    int fileCount = load_pix_files("./pix", files);

    int selected = 0;

    SetTargetFPS(60);

    while (!WindowShouldClose())
    {
        // Input
        if (IsKeyPressed(KEY_DOWN)) selected++;
        if (IsKeyPressed(KEY_UP)) selected--;

        if (selected < 0) selected = 0;
        if (selected >= fileCount) selected = fileCount - 1;

        // Disegno
        BeginDrawing();
        ClearBackground(RAYWHITE);

        DrawText("Seleziona un file .pix:", 20, 20, 20, DARKGRAY);

        for (int i = 0; i < fileCount; i++) {
            Color color = (i == selected) ? RED : BLACK;
            DrawText(files[i], 40, 60 + i * 25, 20, color);
        }

        EndDrawing();
    }

    CloseWindow();
    return 0;
}

🔹 Concetto

Useremo:

  • scrollOffset → indice del primo elemento visibile

  • visibleItems → quanti file entrano nello schermo

  • selected → elemento selezionato

  • rettangoli cliccabili per ogni riga

🔹 Codice completo (scroll + mouse)

C
#include "raylib.h"
#include <dirent.h>
#include <string.h>
#include <stdio.h>

#define MAX_FILES 512
#define MAX_NAME 256

int load_pix_files(const char *path, char files[MAX_FILES][MAX_NAME]) {
    DIR *dir = opendir(path);
    if (!dir) return 0;

    struct dirent *entry;
    int count = 0;

    while ((entry = readdir(dir)) != NULL && count < MAX_FILES) {
        const char *name = entry->d_name;
        const char *ext = strrchr(name, '.');

        if (ext && strcmp(ext, ".pix") == 0) {
            snprintf(files[count], MAX_NAME, "%s", name);
            count++;
        }
    }

    closedir(dir);
    return count;
}

int main(void)
{
    InitWindow(600, 400, "File .pix browser");

    char files[MAX_FILES][MAX_NAME];
    int fileCount = load_pix_files("./pix", files);

    int selected = 0;
    int scrollOffset = 0;

    const int itemHeight = 28;
    const int listTop = 60;
    const int listHeight = 300;

    SetTargetFPS(60);

    while (!WindowShouldClose())
    {
        int visibleItems = listHeight / itemHeight;

        // 🔹 Scroll con mouse wheel
        float wheel = GetMouseWheelMove();
        if (wheel < 0) scrollOffset++;
        if (wheel > 0) scrollOffset--;

        // 🔹 Scroll con tastiera
        if (IsKeyPressed(KEY_DOWN)) selected++;
        if (IsKeyPressed(KEY_UP)) selected--;

        // Clamp selezione
        if (selected < 0) selected = 0;
        if (selected >= fileCount) selected = fileCount - 1;

        // Mantieni selezione visibile
        if (selected < scrollOffset) scrollOffset = selected;
        if (selected >= scrollOffset + visibleItems)
            scrollOffset = selected - visibleItems + 1;

        // Clamp scroll
        if (scrollOffset < 0) scrollOffset = 0;
        if (scrollOffset > fileCount - visibleItems)
            scrollOffset = fileCount - visibleItems;

        if (scrollOffset < 0) scrollOffset = 0;

        // 🔹 Click mouse
        if (IsMouseButtonPressed(MOUSE_LEFT_BUTTON)) {
            Vector2 mouse = GetMousePosition();

            for (int i = 0; i < visibleItems; i++) {
                int index = i + scrollOffset;
                if (index >= fileCount) break;

                Rectangle rect = {
                    20,
                    listTop + i * itemHeight,
                    560,
                    itemHeight
                };

                if (CheckCollisionPointRec(mouse, rect)) {
                    selected = index;
                }
            }
        }

        // 🔹 Disegno
        BeginDrawing();
        ClearBackground(RAYWHITE);

        DrawText("File .pix:", 20, 20, 20, DARKGRAY);

        for (int i = 0; i < visibleItems; i++) {
            int index = i + scrollOffset;
            if (index >= fileCount) break;

            int y = listTop + i * itemHeight;

            Rectangle rect = {20, y, 560, itemHeight};

            // Evidenzia selezionato
            if (index == selected) {
                DrawRectangleRec(rect, LIGHTGRAY);
            }

            // Hover mouse
            Vector2 mouse = GetMousePosition();
            if (CheckCollisionPointRec(mouse, rect)) {
                DrawRectangleLinesEx(rect, 2, RED);
            }

            DrawText(files[index], 30, y + 5, 20, BLACK);
        }

        EndDrawing();
    }

    CloseWindow();
    return 0;
}

🔹 1. Ordinamento alfabetico (qsort)

Dopo aver caricato i file, basta ordinarli:

➤ Comparatore

C
int compare_files(const void *a, const void *b) {
    const char *fa = (const char *)a;
    const char *fb = (const char *)b;
    return strcmp(fa, fb);
}

➤ Uso

Subito dopo load_pix_files:

C
qsort(files, fileCount, MAX_NAME, compare_files);

💡 Nota: così è case-sensitive.
Se vuoi ignorare maiuscole/minuscole:

C
#include <strings.h> // per strcasecmp

int compare_files(const void *a, const void *b) {
    return strcasecmp((const char *)a, (const char *)b);
}

🔹 2. Doppio click del mouse

Raylib non ha una funzione built-in per il double click, quindi lo gestiamo noi con il tempo.

➤ Variabili da aggiungere

C
float lastClickTime = 0.0f;
int lastClickedIndex = -1;
const float doubleClickDelay = 0.3f; // secondi

➤ Logica del click (sostituisci la parte mouse)

C
if (IsMouseButtonPressed(MOUSE_LEFT_BUTTON)) {
    Vector2 mouse = GetMousePosition();
    float currentTime = GetTime();

    for (int i = 0; i < visibleItems; i++) {
        int index = i + scrollOffset;
        if (index >= fileCount) break;

        Rectangle rect = {
            20,
            listTop + i * itemHeight,
            560,
            itemHeight
        };

        if (CheckCollisionPointRec(mouse, rect)) {
            selected = index;

            // 🔹 Check doppio click
            if (index == lastClickedIndex &&
                (currentTime - lastClickTime) < doubleClickDelay) {

                // 👉 QUI "apri" il file
                printf("Apro file: %s\n", files[index]);

                // esempio: costruisci path completo
                char fullpath[512];
                snprintf(fullpath, sizeof(fullpath), "./pix/%s", files[index]);

                // qui potrai caricare il file .pix
            }

            lastClickedIndex = index;
            lastClickTime = currentTime;
        }
    }
}
